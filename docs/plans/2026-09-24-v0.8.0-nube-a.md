# proyecto-loop v0.8.0 — Subproyecto A: el loop en la nube — Plan de implementación

> **Para ejecución agéntica:** usar `superpowers:subagent-driven-development` (recomendado)
> o `superpowers:executing-plans` para recorrer este plan tarea por tarea. Los pasos usan
> checkbox (`- [ ]`).

**Goal:** que un proyecto nazca de una plantilla en GitHub con el loop adentro, y que el
loop corra entero en una sesión cloud sin depender de la máquina local.

**Architecture:** los plugins no cargan en sesiones cloud; lo que llega es lo commiteado en
el repo. Un script, `scripts/vendorear.py`, copia el loop al `.claude/` de un proyecto
reescribiendo lo que solo tiene sentido dentro de un plugin. Una Action lo corre en cada
tag para regenerar un repo plantilla. Las etapas se adaptan a un entorno sin dispositivos
(deuda explícita `[d]`) y sin skills de terceros (registro `NO VERIFICADO`).

**Tech Stack:** Python 3.9+ (solo biblioteca estándar), pytest vía `uv`, skills en
Markdown, GitHub Actions, `gh` CLI.

**Spec:** `docs/specs/2026-09-24-v0.8.0-nube-a-design.md`

## Global Constraints

- **Idioma de todos los prompts de skills y de la prosa: español.**
- El repo `proyecto-loop` es **público**: no commitear nombres internos de repos, del
  servidor de deploy ni ids de dispositivos. Antes de cualquier push corre
  `uv run python scripts/verificar-publicacion.py` y tiene que dar `sin nombres internos: OK`.
- **Las skills de terceros nunca se commitean al repo `proyecto-loop`.** Se bajan desde su
  repo de origen al momento de vendorear, directo al proyecto o a la plantilla.
- Una etapa que otra skill **encadena** nunca lleva `disable-model-invocation: true`. El
  flag es solo para puntos de entrada: `project-loop`, `loop-adopt`, `loop-status`,
  `loop-ship`, `loop-pitch`, y ahora **`loop-actualizar`**.
- **Todo chequeo distingue "verifiqué y está limpio" de "no pude verificar". El segundo
  nunca aprueba.** No colapses códigos de salida con `&&`/`||`, no trates una lista vacía
  como éxito, y escribí las verificaciones para que muestren lo que hay.
- **`vacio` sin tilde es el nombre canónico de un estado de pantalla** (con `default`,
  `carga`, `error`); `vacío` con tilde es la palabra española normal. No unificarlas.
- **`vendorear.py` y `verificar-vendoreo.py` usan solo biblioteca estándar y corren en
  Python 3.9**: empiezan con `from __future__ import annotations`.
- **No manipular credenciales.** El token de la Action lo crea y guarda el usuario.
- Cada tarea termina en un **commit atómico**, con el trailer
  `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```python
#!/usr/bin/env python3
"""Vendorea el loop en un proyecto: arma su .claude/ a partir del plugin.

    uv run python scripts/vendorear.py --destino <ruta-del-proyecto>

Los plugins no cargan en sesiones cloud: lo que llega es lo commiteado en el
repo. Por eso el loop se copia al .claude/ de cada proyecto. Este script es la
única transformación entre el plugin y un proyecto; lo usan la Action que
regenera la plantilla y la skill loop-actualizar.

Solo gestiona lo suyo: lo que quedó listado en .claude/loop/MANIFIESTO. Las
reglas que instaló loop-rules, las skills propias del proyecto y las claves
ajenas de .claude/settings.json no se tocan.
"""
from __future__ import annotations

import argparse
import json
import re
import shutil
import sys
from pathlib import Path

RAIZ = Path(__file__).resolve().parent.parent

# ${CLAUDE_PLUGIN_ROOT} apunta a la raíz de CADA plugin, así que se reescribe
# distinto según de qué plugin venga el archivo.
PREFIJO_LOOP = ".claude/loop"
PREFIJO_SUBSTRATE = ".claude/loop/plugins/dev-substrate"
FUENTES = (
    ("skills", None, PREFIJO_LOOP),
    ("plugins/dev-substrate/skills", "plugins/dev-substrate/agents", PREFIJO_SUBSTRATE),
)
# Recursos no ejecutables: .claude/loop/ es un espejo de su ubicación en el plugin.
RECURSOS = (
    "templates",
    "references",
    "plugins/dev-substrate/rules",
    "plugins/dev-substrate/presets",
    "plugins/dev-substrate/templates",
)
PLUGINS_GLOBALES = (
    "proyecto-loop@proyecto-loop-local",
    "dev-substrate@proyecto-loop-local",
)
# Solo los namespaces propios: los de otros plugins (engineering:, design:) quedan.
NAMESPACE_PROPIO = re.compile(r"\b(?:proyecto-loop|dev-substrate):(?=[a-z])")
TEXTO = {".md", ".txt", ".json", ".yaml", ".yml"}


def reescribir(texto: str, prefijo: str) -> str:
    """Adapta un archivo del plugin para que viva dentro de un proyecto."""
    return NAMESPACE_PROPIO.sub("", texto.replace("${CLAUDE_PLUGIN_ROOT}", prefijo))


def prefijo_de(ruta_relativa: str) -> str:
    if ruta_relativa.startswith("plugins/dev-substrate"):
        return PREFIJO_SUBSTRATE
    return PREFIJO_LOOP


def copiar(origen: Path, destino: Path, prefijo: str) -> None:
    """Copia un árbol reescribiendo los archivos de texto; el resto, byte a byte."""
    for f in sorted(origen.rglob("*")):
        if not f.is_file():
            continue
        dst = destino / f.relative_to(origen)
        dst.parent.mkdir(parents=True, exist_ok=True)
        if f.suffix in TEXTO:
            dst.write_text(reescribir(f.read_text(), prefijo))
        else:
            shutil.copy2(f, dst)


def leer_version(raiz: Path) -> str:
    return json.loads((raiz / ".claude-plugin/plugin.json").read_text())["version"]


def planificar(raiz: Path) -> list[tuple[Path, str, str]]:
    """Qué skills y agentes van a dónde: (origen, destino relativo, prefijo)."""
    plan: list[tuple[Path, str, str]] = []
    vistos: set[str] = set()
    for skills, agentes, prefijo in FUENTES:
        for d in sorted((raiz / skills).iterdir()):
            if not (d / "SKILL.md").is_file():
                continue
            if d.name in vistos:
                raise SystemExit(f"skill duplicada entre plugins: {d.name}")
            vistos.add(d.name)
            plan.append((d, f".claude/skills/{d.name}", prefijo))
        if agentes:
            for a in sorted((raiz / agentes).glob("*.md")):
                plan.append((a, f".claude/agents/{a.name}", prefijo))
    return plan


def desactivar_plugins_globales(destino: Path) -> None:
    """Que en local corra la copia vendoreada y no la del plugin global."""
    p = destino / ".claude/settings.json"
    datos = json.loads(p.read_text()) if p.exists() else {}
    habilitados = datos.setdefault("enabledPlugins", {})
    for plugin in PLUGINS_GLOBALES:
        habilitados[plugin] = False
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(datos, indent=2, ensure_ascii=False) + "\n")


def vendorear(raiz: Path, destino: Path) -> list[str]:
    """Arma o actualiza el .claude/ de `destino`. Devuelve las rutas gestionadas."""
    loop = destino / PREFIJO_LOOP
    manifiesto = loop / "MANIFIESTO"
    previas = set(manifiesto.read_text().split()) if manifiesto.exists() else set()
    plan = planificar(raiz)

    # Antes de borrar nada: no pisar lo que el proyecto creó por su cuenta.
    for _, rel, _ in plan:
        if (destino / rel).exists() and rel not in previas:
            raise SystemExit(f"{rel} ya existe y no es del loop: no se pisa")

    for rel in previas:
        p = destino / rel
        if p.is_dir():
            shutil.rmtree(p)
        elif p.exists():
            p.unlink()
    if loop.exists():
        shutil.rmtree(loop)

    gestionadas: list[str] = []
    for origen, rel, prefijo in plan:
        if origen.is_dir():
            copiar(origen, destino / rel, prefijo)
        else:
            dst = destino / rel
            dst.parent.mkdir(parents=True, exist_ok=True)
            dst.write_text(reescribir(origen.read_text(), prefijo))
        gestionadas.append(rel)

    for r in RECURSOS:
        copiar(raiz / r, loop / r, prefijo_de(r))

    desactivar_plugins_globales(destino)
    loop.mkdir(parents=True, exist_ok=True)
    (loop / "VERSION").write_text(leer_version(raiz) + "\n")
    manifiesto.write_text("\n".join(gestionadas) + "\n")
    return gestionadas


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="Vendorea el loop en un proyecto.")
    ap.add_argument("--destino", required=True, type=Path, help="raíz del proyecto")
    ap.add_argument("--origen", type=Path, default=RAIZ, help="raíz de proyecto-loop")
    a = ap.parse_args(argv)
    rutas = vendorear(a.origen.resolve(), a.destino.resolve())
    print(f"vendoreado: {len(rutas)} skills y agentes · versión "
          f"{leer_version(a.origen)} → {a.destino}/.claude")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def obtener_tercero(url: str, sha: str, cache: Path) -> Path:
    """Trae el repo de origen en un commit exacto: por SHA, nunca por rama."""
    d = cache / sha
    if not (d / ".git").is_dir():
        d.mkdir(parents=True, exist_ok=True)
        for cmd in (
            ["git", "init", "-q"],
            ["git", "fetch", "-q", "--depth", "1", url, sha],
            ["git", "checkout", "-q", "--detach", "FETCH_HEAD"],
        ):
            subprocess.run(cmd, cwd=d, check=True)
    return d


def nombre_declarado(skill_md: Path) -> str | None:
    m = re.search(r"^name:\s*(\S+)", leer_texto(skill_md), re.M)
    return m.group(1) if m else None


def planificar_terceros(terceros: dict, cache: Path) -> list[tuple[Path, str, dict]]:
    """Trae cada skill de terceros y confirma que declare el nombre esperado."""
    plan: list[tuple[Path, str, dict]] = []
    for nombre, t in sorted(terceros.items()):
        repo = obtener_tercero(t["url"], t["sha"], cache)
        real = nombre_declarado(repo / t["ruta"] / "SKILL.md")
        if real != nombre:
            raise SystemExit(f"{nombre}: la skill en {t['ruta']} declara name={real!r}")
        if not (repo / "LICENSE").is_file():
            raise SystemExit(f"{nombre}: el repo de origen no tiene LICENSE")
        plan.append((repo, f".claude/skills/{nombre}", t))
    return plan


def escribir_terceros_md(loop: Path, plan: list[tuple[Path, str, dict]]) -> None:
    lineas = ["# Skills de terceros vendoreadas", ""]
    if not plan:
        lineas.append("Ninguna. Este vendoreo se hizo sin skills de terceros.")
    else:
        lineas += ["Copiadas sin modificar desde su repo de origen, fijadas por commit.",
                   "Cada una trae su `LICENSE`.", "",
                   "| Skill | Origen | Commit | Licencia |", "|---|---|---|---|"]
        for _, rel, t in plan:
            lineas.append(f"| `{rel.rsplit('/', 1)[1]}` | {t['url']} | "
                          f"`{t['sha']}` | {t['licencia']} |")
    (loop / "TERCEROS.md").write_text("\n".join(lineas) + "\n", encoding="utf-8")
```

```python
#!/usr/bin/env python3
"""Verifica que el .claude/ de un proyecto sea un vendoreo correcto del loop.

    uv run python scripts/verificar-vendoreo.py --destino <proyecto> [--origen <proyecto-loop>]

Lo corren la Action que regenera la plantilla y la skill loop-actualizar antes
de commitear. Salida: 0 = verificado · 1 = hay fallas, o no se pudo verificar.
"""
from __future__ import annotations

import argparse
import json
import re
import sys
from pathlib import Path

RAIZ = Path(__file__).resolve().parent.parent
PLUGINS_GLOBALES = ("proyecto-loop@proyecto-loop-local", "dev-substrate@proyecto-loop-local")
RUTA_LOOP = re.compile(r"\.claude/loop/[\w./-]+")
NAMESPACE_PROPIO = re.compile(r"\b(?:proyecto-loop|dev-substrate):(?=[a-z])")


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="Verifica un vendoreo del loop.")
    ap.add_argument("--destino", required=True, type=Path)
    ap.add_argument("--origen", type=Path, default=RAIZ)
    ap.add_argument("--sin-terceros", action="store_true",
                    help="el vendoreo se hizo sin skills de terceros")
    a = ap.parse_args(argv)
    d, o = a.destino.resolve(), a.origen.resolve()
    claude = d / ".claude"
    manifiesto = claude / "loop/MANIFIESTO"
    if not manifiesto.is_file():
        print(f"AVISO: no existe {manifiesto}: no hay un vendoreo que verificar.")
        return 1

    errores: list[str] = []
    gestionadas = manifiesto.read_text().split()
    terceros = {} if a.sin_terceros else json.loads((o / "terceros.json").read_text())
    de_terceros = {f".claude/skills/{n}" for n in terceros}

    version = json.loads((o / ".claude-plugin/plugin.json").read_text())["version"]
    archivo_version = claude / "loop/VERSION"
    vendoreada = archivo_version.read_text().strip() if archivo_version.is_file() else None
    if vendoreada != version:
        errores.append(f"VERSION dice {vendoreada!r} y el plugin {version!r}")

    for rel in gestionadas:
        if not (d / rel).exists():
            errores.append(f"el MANIFIESTO lista {rel} y no existe")

    # Lo propio del loop. El código de terceros no se revisa: no se reescribe.
    archivos: list[Path] = []
    for rel in gestionadas:
        if rel in de_terceros:
            continue
        p = d / rel
        archivos += [p] if p.is_file() else [f for f in p.rglob("*.md") if f.is_file()]
    for f in archivos:
        texto = f.read_text()
        nombre = f.relative_to(d)
        if "${CLAUDE_PLUGIN_ROOT}" in texto:
            errores.append(f"{nombre}: quedó CLAUDE_PLUGIN_ROOT sin reescribir")
        if NAMESPACE_PROPIO.search(texto):
            errores.append(f"{nombre}: quedó un namespace propio sin quitar")
        for ruta in RUTA_LOOP.findall(texto):
            ruta = ruta.rstrip("/.,;:)`")
            if not (d / ruta).exists():
                errores.append(f"{nombre}: cita {ruta}, que no existe")
    if not archivos:
        errores.append("no se revisó ningún archivo propio del loop: el vendoreo está vacío")

    for n in terceros:
        s = claude / "skills" / n
        if not (s / "SKILL.md").is_file():
            errores.append(f"falta la skill de terceros {n}")
        elif not (s / "LICENSE").is_file():
            errores.append(f"la skill de terceros {n} no trae su LICENSE")

    ajustes = claude / "settings.json"
    habilitados = json.loads(ajustes.read_text()).get("enabledPlugins", {}) if ajustes.is_file() else {}
    for p in PLUGINS_GLOBALES:
        if habilitados.get(p) is not False:
            errores.append(f"settings.json no desactiva el plugin global {p}")

    if errores:
        print("FALLA:")
        for e in errores:
            print("  -", e)
        return 1
    print(f"vendoreo: OK · versión {version} · {len(gestionadas)} rutas · "
          f"{len(archivos)} archivos revisados · {len(terceros)} de terceros")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
